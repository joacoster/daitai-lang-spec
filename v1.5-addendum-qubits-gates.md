## §16.7 Qubits and Quantum Gates

### Primitive Qubit Type

```daitai
sort Qubit

-- Basis states
ket zero: Qubit = |0⟩
ket one: Qubit  = |1⟩

-- Superposition (from basis)
function superposition(alpha: Complex, beta: Complex) -> Qubit:
    return alpha * |0⟩ + beta * |1⟩

-- Single-Qubit Gates

-- Pauli gates
operator X : Qubit → Qubit    -- NOT gate: |0⟩↔|1⟩
operator Y : Qubit → Qubit    -- Pauli-Y: iσ_y
operator Z : Qubit → Qubit    -- Pauli-Z: phase flip

-- Hadamard (creates superposition)
operator H : Qubit → Qubit
-- H|0⟩ = |+⟩ = (|0⟩+|1⟩)/√2
-- H|1⟩ = |−⟩ = (|0⟩-|1⟩)/√2

-- Phase gates
operator S : Qubit → Qubit    -- Phase: diag(1, i)
operator T : Qubit → Qubit    -- T: diag(1, e^{iπ/4})

-- Rotation gates
operator RX(θ: Float) : Qubit → Qubit
operator RY(θ: Float) : Qubit → Qubit
operator RZ(θ: Float) : Qubit → Qubit

Multi-Qubit Gates

-- Controlled-NOT: |c⟩|t⟩ → |c⟩|c ⊕ t⟩
operator CNOT : (Qubit, Qubit) → (Qubit, Qubit)

-- Controlled-Z
operator CZ : (Qubit, Qubit) → (Qubit, Qubit)

-- SWAP
operator SWAP : (Qubit, Qubit) → (Qubit, Qubit)

-- Toffoli (CCNOT)
operator Toffoli : (Qubit, Qubit, Qubit) → (Qubit, Qubit, Qubit)

-- Fredkin (CSWAP)
operator Fredkin : (Qubit, Qubit, Qubit) → (Qubit, Qubit, Qubit)

Quantum Circuits

-- A circuit is a sequence of gates (unitary operations)
class QuantumCircuit:
    qubits: List<Qubit>
    gates: List<Operator>

    method append(gate: Operator) -> QuantumCircuit:
        return QuantumCircuit(qubits = this.qubits, gates = this.gates.append(gate))

    method then(other: QuantumCircuit) -> QuantumCircuit:
        -- Sequential composition
        return QuantumCircuit(
            qubits = this.qubits,
            gates = this.gates.appendAll(other.gates)
        )

    method tensor(other: QuantumCircuit) -> QuantumCircuit:
        -- Parallel composition (⊗)
        return QuantumCircuit(
            qubits = this.qubits.appendAll(other.qubits),
            gates = this.gates.appendAll(other.gates)
        )

Measurement

-- Measurement collapses the quantum state (explicitly nondeterministic)
@nondeterministic
operator measure : Qubit → Bit

-- Measurement in a basis (e.g., X-basis, Z-basis)
@nondeterministic
operator measureBasis(qubit: Qubit, basis: Basis) -> Bit

enum Basis:
    PauliZ   -- Standard basis (|0⟩, |1⟩)
    PauliX   -- Hadamard basis (|+⟩, |−⟩)
    PauliY   -- Circular basis

No-Cloning Theorem (Constitutional Rule)

-- The following operation is CONSTITUTIONALLY FORBIDDEN:
-- operator clone : Qubit → (Qubit, Qubit)

-- Reason: No-cloning theorem prohibits copying an unknown quantum state.
-- This is enforced by the daitai constitution (see §3).

Example: Bell State (Entanglement)

function createBellState() -> QuantumCircuit:
    val q0: Qubit = |0⟩
    val q1: Qubit = |0⟩
    
    return
        H(q0)
        then CNOT(q0, q1)
        -- Result: (|00⟩ + |11⟩)/√2

@nondeterministic
function measureBellState(circuit: QuantumCircuit) -> (Bit, Bit):
    val b0 = measure(circuit.q0)
    val b1 = measure(circuit.q1)
    return (b0, b1)
    -- Always same outcome: (0,0) or (1,1) — perfect correlation
